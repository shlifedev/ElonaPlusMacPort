cmake_minimum_required(VERSION 3.24)
project(ElonaPlatformTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_LIST_DIR}/../../src/cmake/ElonaPlatform.cmake")
enable_testing()
foreach(name test_encoding test_game_files)
  add_executable(${name} "${CMAKE_CURRENT_LIST_DIR}/${name}.cpp")
  target_link_libraries(${name} PRIVATE elona_platform)
  if(MSVC)
    target_compile_options(${name} PRIVATE /utf-8)
  endif()
endforeach()
set(ELONA_CP932_TABLE_DIR "" CACHE PATH "Directory with Windows cp932-decode.tsv/cp932-encode.tsv (optional full-table test)")
add_test(NAME platform_encoding COMMAND test_encoding "${ELONA_CP932_TABLE_DIR}")
add_test(NAME platform_game_files COMMAND test_game_files "${CMAKE_CURRENT_BINARY_DIR}/fs-scratch")
if(TARGET elona_platform_hsp)
  add_executable(test_game_files_hsp "${CMAKE_CURRENT_LIST_DIR}/test_game_files_hsp.cpp" "${ELONA_PLATFORM_HSP3_DIR}/strbuf.cpp")
  target_link_libraries(test_game_files_hsp PRIVATE elona_platform_hsp)
  target_include_directories(test_game_files_hsp PRIVATE "${ELONA_PLATFORM_HSP3_DIR}")
  get_target_property(_hsp_defs elona_platform_hsp COMPILE_DEFINITIONS)
  target_compile_definitions(test_game_files_hsp PRIVATE ${_hsp_defs})
  if(MSVC)
    target_compile_options(test_game_files_hsp PRIVATE /utf-8)
  else()
    target_compile_options(test_game_files_hsp PRIVATE -Wno-write-strings)
  endif()
  add_test(NAME platform_game_files_hsp COMMAND test_game_files_hsp "${CMAKE_CURRENT_BINARY_DIR}/hsp-scratch")
endif()

# ---- End-to-end: real HSP VM + compiler built here (same sources/flags as the
# root legacy-CP932 preset), independent of other build directories.
option(ELONA_PLATFORM_E2E "Build the HSP VM end-to-end file-command test" ON)
set(OPENHSP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.cache/upstream/OpenHSP" CACHE PATH "Pinned OpenHSP checkout")
if(ELONA_PLATFORM_E2E AND TARGET elona_platform_hsp AND EXISTS "${OPENHSP_ROOT}/src/hsp3/hsp3.cpp")
  set(_root "${CMAKE_CURRENT_LIST_DIR}/../..")
  set(_hsp3 "${OPENHSP_ROOT}/src/hsp3")
  set(_cmp "${OPENHSP_ROOT}/src/hspcmp")
  if(WIN32)
    set(_supio "${_hsp3}/win32gui/supio_win.cpp")
    set(_plat HSPWIN)
  else()
    set(_supio "${_hsp3}/linux/supio_linux.cpp")
    set(_plat HSPLINUX)
  endif()
  set(_core)
  foreach(n hsp3 hsp3code hsp3debug hsp3int hspvar_core hspvar_double hspvar_int hspvar_label hspvar_str hspvar_struct stack strbuf strnote dpmread filepack hsp3crypt hsp3utfcnv)
    list(APPEND _core "${_hsp3}/${n}.cpp")
  endforeach()
  add_library(platform_e2e_core STATIC ${_core} ${_supio} "${_root}/src/legacy_logical_path.cpp" "${_root}/src/legacy_rng.cpp")
  target_include_directories(platform_e2e_core PUBLIC "${_hsp3}" PRIVATE "${_root}/src")
  target_compile_definitions(platform_e2e_core PUBLIC HSPDEBUG HSP64 HSP_COM_UNSUPPORTED PRIVATE ELONA_LEGACY_CP932 ELONA_LEGACY_RNG)
  if(WIN32)
    target_compile_definitions(platform_e2e_core PUBLIC HSPWIN)
    target_link_libraries(platform_e2e_core PUBLIC shell32 user32 winmm ole32 oleaut32)
  elseif(APPLE)
    target_compile_definitions(platform_e2e_core PUBLIC HSPMAC)
  else()
    target_compile_definitions(platform_e2e_core PUBLIC HSPLINUX)
  endif()
  set(_cmpsrc)
  foreach(n ahtmodel ahtobj codegen comutil errormsg hsc3 hspcmd label localinfo main membuf tagstack hsmanager token)
    list(APPEND _cmpsrc "${_cmp}/${n}.cpp")
  endforeach()
  add_executable(platform_e2e_hspcmp ${_cmpsrc} "${_hsp3}/strnote.cpp" "${_hsp3}/strbuf.cpp" ${_supio} "${_root}/src/legacy_logical_path.cpp")
  target_include_directories(platform_e2e_hspcmp PRIVATE "${_root}/src")
  target_compile_definitions(platform_e2e_hspcmp PRIVATE ${_plat} HSPDEBUG HSP64 HSP_COM_UNSUPPORTED ELONA_LEGACY_CP932)
  add_executable(platform_hsp_file_host "${CMAKE_CURRENT_LIST_DIR}/hsp_file_host.cpp")
  target_link_libraries(platform_hsp_file_host PRIVATE elona_platform_hsp platform_e2e_core)
  foreach(t platform_e2e_core platform_e2e_hspcmp platform_hsp_file_host)
    if(MSVC)
      target_compile_options(${t} PRIVATE /utf-8)
      target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    else()
      target_compile_options(${t} PRIVATE -Wno-write-strings -w)
    endif()
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME platform_hsp_e2e COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/run_hsp_e2e.py"
      --compiler "$<TARGET_FILE:platform_e2e_hspcmp>" --host "$<TARGET_FILE:platform_hsp_file_host>"
      --common "${OPENHSP_ROOT}/common" --work "${CMAKE_CURRENT_BINARY_DIR}/e2e")
  endif()
endif()
