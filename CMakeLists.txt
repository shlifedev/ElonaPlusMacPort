cmake_minimum_required(VERSION 3.24)
project(ElonaPlusNative LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This diagnostic spike targets native 64-bit hosts only")
endif()
option(ELONA_LEGACY_CP932 "Opt in to CP932-aware POSIX string walkers (not full game compatibility)" OFF)
option(ELONA_LEGACY_RNG "Use the HSP 3.4 Windows reference RNG sequence instead of host libc rand" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENHSP_ROOT "${CMAKE_SOURCE_DIR}/.cache/upstream/OpenHSP" CACHE PATH "Pinned OpenHSP checkout")
if(NOT EXISTS "${OPENHSP_ROOT}/src/hsp3/hsp3.cpp")
  message(FATAL_ERROR "Run python3 tools/bootstrap_openhsp.py first")
endif()
set(HSP3 "${OPENHSP_ROOT}/src/hsp3")
set(HSPCMP "${OPENHSP_ROOT}/src/hspcmp")
if(WIN32)
  set(HSP_SUPIO "${HSP3}/win32gui/supio_win.cpp")
  set(HSP_COMPILER_PLATFORM HSPWIN)
else()
  set(HSP_SUPIO "${HSP3}/linux/supio_linux.cpp")
  set(HSP_COMPILER_PLATFORM HSPLINUX)
endif()
set(COMPILER_FILES ahtmodel ahtobj codegen comutil errormsg hsc3 hspcmd label localinfo main membuf tagstack hsmanager token)
set(COMPILER_SOURCES)
foreach(name IN LISTS COMPILER_FILES)
  list(APPEND COMPILER_SOURCES "${HSPCMP}/${name}.cpp")
endforeach()
add_executable(elona_hspcmp ${COMPILER_SOURCES} "${HSP3}/strnote.cpp" "${HSP3}/strbuf.cpp" ${HSP_SUPIO})
target_compile_definitions(elona_hspcmp PRIVATE ${HSP_COMPILER_PLATFORM} HSPDEBUG HSP64 HSP_COM_UNSUPPORTED)

set(CORE_FILES hsp3 hsp3code hsp3debug hsp3int hspvar_core hspvar_double hspvar_int hspvar_label hspvar_str hspvar_struct stack strbuf strnote dpmread filepack hsp3crypt hsp3utfcnv)
set(CORE_SOURCES)
foreach(name IN LISTS CORE_FILES)
  list(APPEND CORE_SOURCES "${HSP3}/${name}.cpp")
endforeach()
add_library(elona_hsp_core STATIC ${CORE_SOURCES} ${HSP_SUPIO})
target_include_directories(elona_hsp_core PUBLIC "${HSP3}")
target_compile_definitions(elona_hsp_core PUBLIC HSPDEBUG HSP64 HSP_COM_UNSUPPORTED)
if(WIN32)
  target_compile_definitions(elona_hsp_core PUBLIC HSPWIN)
elseif(APPLE)
  target_compile_definitions(elona_hsp_core PUBLIC HSPMAC)
else()
  target_compile_definitions(elona_hsp_core PUBLIC HSPLINUX)
endif()
add_executable(elona_hsp_probe src/native_probe.cpp)
target_link_libraries(elona_hsp_probe PRIVATE elona_hsp_core)
foreach(target elona_hspcmp elona_hsp_core elona_hsp_probe)
  if(ELONA_LEGACY_CP932)
    target_compile_definitions(${target} PRIVATE ELONA_LEGACY_CP932)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  else()
    target_compile_options(${target} PRIVATE -Wno-write-strings)
  endif()
endforeach()
if(ELONA_LEGACY_CP932)
  foreach(target elona_hspcmp elona_hsp_core)
    target_sources(${target} PRIVATE src/legacy_logical_path.cpp)
    target_include_directories(${target} PRIVATE "${CMAKE_SOURCE_DIR}/src")
  endforeach()
endif()
if(ELONA_LEGACY_RNG)
  target_compile_definitions(elona_hsp_core PRIVATE ELONA_LEGACY_RNG)
  target_sources(elona_hsp_core PRIVATE src/legacy_rng.cpp)
  target_include_directories(elona_hsp_core PRIVATE "${CMAKE_SOURCE_DIR}/src")
endif()
if(WIN32)
  target_link_libraries(elona_hspcmp PRIVATE shell32 user32 winmm ole32 oleaut32)
  target_link_libraries(elona_hsp_core PUBLIC shell32 user32 winmm ole32 oleaut32)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(elona_string_tests tests/native/string_walkers.cpp)
  target_link_libraries(elona_string_tests PRIVATE elona_hsp_core)
  if(MSVC)
    target_compile_options(elona_string_tests PRIVATE /utf-8)
    target_compile_definitions(elona_string_tests PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  endif()
  if(ELONA_LEGACY_CP932 OR WIN32)
    target_compile_definitions(elona_string_tests PRIVATE ELONA_EXPECT_CP932)
  endif()
  add_test(NAME native_string_walkers COMMAND elona_string_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    set(SEMANTIC_FLAGS)
    if(ELONA_LEGACY_CP932 OR WIN32)
      list(APPEND SEMANTIC_FLAGS --legacy-cp932)
    endif()
    if(ELONA_LEGACY_RNG)
      list(APPEND SEMANTIC_FLAGS --legacy-rng)
    endif()
    add_test(NAME native_semantics COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/native/run_semantics.py" --compiler "$<TARGET_FILE:elona_hspcmp>" --runtime "$<TARGET_FILE:elona_hsp_probe>" --common "${OPENHSP_ROOT}/common" --work "${CMAKE_BINARY_DIR}/semantic-tests" ${SEMANTIC_FLAGS})
  else()
    message(WARNING "Python3 not available: native_semantics is not registered; use the separately documented Windows fixture comparison")
  endif()
endif()

# Optional shared modules from other workers (included only when present).
if(EXISTS "${CMAKE_SOURCE_DIR}/cmake/ElonaPlatform.cmake")
  include(cmake/ElonaPlatform.cmake)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/cmake/ElonaPlugins.cmake")
  include(cmake/ElonaPlugins.cmake)
  elona_add_plugins(TESTS ${BUILD_TESTING})
endif()
# Music renderer (MIDI/MP3) when its pinned inputs were fetched with
# tools/bootstrap_audio.py; otherwise the host keeps the documented silent MCI path.
option(ELONA_MUSIC "Build src/audio music renderer when inputs are present" ON)
if(ELONA_MUSIC AND EXISTS "${CMAKE_SOURCE_DIR}/cmake/ElonaAudio.cmake"
   AND EXISTS "${CMAKE_SOURCE_DIR}/.cache/audio/TinySoundFont/tsf.h")
  include(cmake/ElonaAudio.cmake)
  elona_add_audio(TESTS ${BUILD_TESTING})
endif()
# Native pixel model, SDL3 window backend and GUI host.
include(cmake/ElonaGraphics.cmake)
